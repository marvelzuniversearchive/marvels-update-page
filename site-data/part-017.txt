function toonBlock(x,cx,top,w,h,col,o){
  const base=mixHex(col,o.hazeC,o.haze), shadow=mixHex(shadeOf(col,.62),o.hazeC,o.haze);
  const side=w*.17, front=w-side, roof=Math.max(2,Math.min(5,w*.025));
  const face=x.createLinearGradient(cx,top,cx+front,top+h);
  face.addColorStop(0,shadeOf(base,1.08));face.addColorStop(.48,base);face.addColorStop(1,shadeOf(base,.78));
  x.fillStyle=face;x.beginPath();x.moveTo(cx,top+roof);x.lineTo(cx+front,top);x.lineTo(cx+front,top+h);x.lineTo(cx,top+h);x.closePath();x.fill();
  const sideGrad=x.createLinearGradient(cx+front,top,cx+w,top);
  sideGrad.addColorStop(0,shadow);sideGrad.addColorStop(1,shadeOf(shadow,.78));
  x.fillStyle=sideGrad;x.beginPath();x.moveTo(cx+front,top);x.lineTo(cx+w,top+roof);x.lineTo(cx+w,top+h);x.lineTo(cx+front,top+h);x.closePath();x.fill();
  x.fillStyle=shadeOf(base,1.13);x.beginPath();x.moveTo(cx,top+roof);x.lineTo(cx+front,top);x.lineTo(cx+w,top+roof);x.lineTo(cx+w-side*.08,top+roof+2);x.closePath();x.fill();
  x.save();x.globalAlpha=o.dotA*.38;x.fillStyle=o.dots;x.fillRect(cx+front,top+roof,w-front,h-roof);x.restore();
  x.save();x.globalAlpha=.72;x.strokeStyle=o.ink;x.lineWidth=Math.max(.55,o.lw*.38);
  x.beginPath();x.moveTo(cx,top+roof);x.lineTo(cx+front,top);x.lineTo(cx+w,top+roof);x.lineTo(cx+w,top+h);x.lineTo(cx,top+h);x.closePath();x.stroke();

  return side;

      if(lit){glass.addColorStop(0,shadeOf(warm,1.08));glass.addColorStop(.55,warm);glass.addColorStop(1,shadeOf(warm,.58))}

function updateTrails(dt){
  for(let i=trails.length-1;i>=0;i--){
    const tr=trails[i];
    tr.x+=Math.cos(tr.a)*tr.spd*dt;tr.y+=Math.sin(tr.a)*tr.spd*dt;tr.life-=dt;
    if(tr.life<=0||tr.x<-400||tr.x>W+400||tr.y<-400||tr.y>H+400)trails.splice(i,1);
  }
}
function updateParts(dt){
  for(let i=parts.length-1;i>=0;i--){
    const p=parts[i];
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;
    if(p.life!==Infinity){p.life-=dt;if(p.life<=0){parts.splice(i,1);continue}}
    if(p.x<-12)p.x+=W+24;else if(p.x>W+12)p.x-=W+24;
    if(p.y<-12)p.y+=H+24;else if(p.y>H+12)p.y-=H+24;
  }
}

function step(dt){
  let e=0;
  if(!reduced){
    t+=dt;
    ptr.x+=(ptr.tx-ptr.x)*Math.min(1,dt*4);
    ptr.y+=(ptr.ty-ptr.y)*Math.min(1,dt*4);
    scr.y+=(scr.target-scr.y)*Math.min(1,dt*6);
    leapClock+=dt;
    if(leapT<0&&leapClock>=nextLeap){
      leapT=0;nextLeap=leapClock+(isSmall()?22:16)*rnd(.9,1.15);
      for(let i=0;i<(isSmall()?6:14);i++)parts.push(burstPart());
      for(let i=0;i<3;i++)spawnTrail(true);
    }
    if(leapT>=0){leapT+=dt;const u=leapT/LEAP_DUR;if(u>=1)leapT=-1;else e=Math.sin(Math.PI*u)}
    spawnAcc+=(isSmall()?.25:.5)*(1+e*5)*dt;
    while(spawnAcc>=1){spawnAcc-=1;if(trails.length<(isSmall()?6:14))spawnTrail(e>.2)}
    updateClouds(dt);updateTrails(dt);updateParts(dt);
    for(const L of layers)L.off=(L.off+L.speed*(1+e*L.boost)*dt)%L.tw;
  }
  render(e);
}

function loop(now){
  raf=0;
  const minGap=1000/(isSmall()?24:30);

    const dt=Math.min(.05,Math.max(0,(now-last)/1000));last=now;step(dt);
  }
  if(!document.hidden)raf=requestAnimationFrame(loop);
}

  last=performance.now();raf=requestAnimationFrame(loop);
}

  if(!glFx||!glBadge||!glText)return;
  glFx.classList.remove('gl-run','gl-fade');
  void glFx.offsetWidth;
  glFx.classList.add('gl-run');
  glText.textContent='INITIALIZING MARVEL ARCHIVE...';
  glBadge.classList.add('on');
  if(reduced){
    glFx.classList.add('gl-fade');
    setTimeout(()=>{glFx.classList.remove('gl-run','gl-fade');glBadge.classList.remove('on')},1150);
    return;
  }
  root.classList.add('glitching');
  burstFx();
  setTimeout(()=>{
    glFx.classList.remove('gl-run','gl-fade');
    root.classList.remove('glitching');
    glBadge.classList.remove('on');
    glEnd=performance.now();
  },1250);
}
